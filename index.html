import { createClient } from "npm:@supabase/supabase-js@2"; const corsHeaders = { "Access-Control-Allow-Origin": "https://peace-d27.github.io", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "GET, POST, OPTIONS", }; const respond = (data: unknown, status = 200) => new Response(JSON.stringify(data), { status, headers: { ...corsHeaders, "Content-Type": "application/json", "Cache-Control": "no-store", }, }); const maxLength = (value: unknown, n: number) => typeof value === "string" && value.trim().length > 0 && value.length <= n; Deno.serve(async (req) => { // Handle CORS preflight requests if (req.method === "OPTIONS") { return new Response(null, { status: 204, headers: corsHeaders, }); } // Connect to Supabase const supabaseUrl = Deno.env.get("SUPABASE_URL"); const serviceRoleKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY"); if (!supabaseUrl || !serviceRoleKey) { return respond( { error: "Server configuration missing" }, 500 ); } const db = createClient(supabaseUrl, serviceRoleKey); // GET: Load published Peace Wall contributions if (req.method === "GET") { const { data, error } = await db .from("peace_contributions") .select( "id,created_at,card_number,response_text,attribution_type,first_name,last_initial,city" ) .eq("moderation_status", "published") .eq("publication_consent", true) .order("created_at", { ascending: false }) .limit(100); if (error) { console.error("Wall loading failed", error.message); return respond({ error: "Wall unavailable" }, 500); } return respond({ contributions: data }); } // Only GET, POST and OPTIONS are allowed if (req.method !== "POST") { return respond({ error: "Method not allowed" }, 405); } // Read and validate JSON request let body: Record; try { const parsed: unknown = await req.json(); if ( parsed === null || typeof parsed !== "object" || Array.isArray(parsed) ) { return respond({ error: "Invalid JSON" }, 400); } body = parsed as Record; } catch { return respond({ error: "Invalid JSON" }, 400); } // REPORT: Report an existing contribution if (body.action === "report") { const contributionId = body.contribution_id; if ( typeof contributionId !== "number" || !Number.isSafeInteger(contributionId) || contributionId <= 0 || !maxLength(body.reason, 300) ) { return respond({ error: "Invalid report" }, 400); } // Check whether the contribution exists const { data: contribution, error: lookupError } = await db .from("peace_contributions") .select("id") .eq("id", contributionId) .maybeSingle(); if (lookupError) { console.error( "Report lookup failed", lookupError.message ); return respond( { error: "Could not verify contribution" }, 500 ); } if (!contribution) { return respond( { error: "Contribution not found" }, 404 ); } // Save report. // peace_reports.contribution_id has type TEXT. const { error } = await db .from("peace_reports") .insert({ contribution_id: String(contributionId), reason: (body.reason as string).trim(), }); if (error) { console.error( "Report insert failed", error.message ); return respond( { error: "Report could not be saved" }, 500 ); } return respond({ success: true }); } // CONTRIBUTION: Validate a new Peace Wall entry const cardNumber = body.card_number; const named = body.attribution_type === "named"; if ( body.publication_consent !== true || !["named", "anonymous"].includes( String(body.attribution_type) ) || ![1, 2, 3].includes(cardNumber as number) || !maxLength(body.response_text, 2000) ) { return respond( { error: "Invalid contribution" }, 400 ); } // Validate named attribution if ( named && ( !maxLength(body.first_name, 80) || !maxLength(body.city, 100) || typeof body.last_initial !== "string" || !/^\p{L}$/u.test(body.last_initial) ) ) { return respond( { error: "Invalid attribution" }, 400 ); } // Publish contribution const { error } = await db .from("peace_contributions") .insert({ card_number: cardNumber, response_text: (body.response_text as string).trim(), attribution_type: named ? "named" : "anonymous", first_name: named ? (body.first_name as string).trim() : null, last_initial: named ? body.last_initial : null, city: named ? (body.city as string).trim() : null, publication_consent: true, moderation_status: "published", }); if (error) { console.error( "Contribution insert failed", error.message ); return respond( { error: "Could not save contribution" }, 500 ); } return respond( { success: true, message: "Published", }, 201 ); });
